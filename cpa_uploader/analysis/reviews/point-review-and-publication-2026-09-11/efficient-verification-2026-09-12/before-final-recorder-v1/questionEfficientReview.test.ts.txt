function fixture(count = 1) {

        const absolute = path.join(root, file); fs.mkdirSync(path.dirname(absolute), { recursive: true });

        grading_manifest: { file: '', sha256: '' }, observations: [], agent_reviews: reviews, runtime_snapshots: runtime, residual_grading_findings: [] };
    function flush() {
        batch.grading_manifest = write('manifest.json', manifest);
        batch.observations = jobs.map(job => {

function syncOutput(job: FixtureJob) {
    job.response.output_text = JSON.stringify(job.raw);
    job.response.output = [{ type: 'message', id: 'msg_fixture', role: 'assistant', status: 'completed', content: [{ type: 'output_text', text: job.response.output_text, annotations: [] }] }];
}
function regradeSynthetic(job: FixtureJob) {
    syncOutput(job); const o = job.observation;
    o.judgment = groundJudgment(job.raw, job.projected, job.entry.answers); o.result = applyQuestionSetJudgment(job.projected, job.entry.answers, o.judgment);
    o.subquestions = o.result.subquestions.map(q => { const e = job.entry.expected_by_subquestion.find(e => e.subquestion_id === q.subquestion_id)!;
        const delta = q.score - e.expected_points; return { subquestion_id: q.subquestion_id, evaluated: job.entry.evaluated_subquestion_ids.includes(q.subquestion_id),
            expected_points: e.expected_points, actual_points: q.score, delta, strict_matched: delta === 0 && e.expected_verdicts.every(v => q.criteria.find(c => c.criterion_id === v.criterion_id)?.verdict === v.verdict), within_tolerance: Math.abs(delta) <= 1 }; });
    o.strict_matched = o.subquestions.every(q => q.strict_matched); o.within_tolerance = o.subquestions.every(q => q.within_tolerance);
}

function resumedFixture(f: Fixture) {
    f.manifest.inputs.push(f.write('origin/runtime-snapshots.json', f.batch.runtime_snapshots));
    f.flush();
    const manifest = structuredClone(f.manifest), batch = structuredClone(f.batch);
